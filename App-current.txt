import React, { useState } from 'react';
import { BrowserRouter, Routes, Route, Navigate } from 'react-router-dom';
import Login from './pages/Login';
import Dashboard from './pages/Dashboard';
import Members from './pages/Members';
import ChitsPlaceholder from './pages/ChitsPlaceholder';
import PaymentsPlaceholder from './pages/PaymentsPlaceholder';
import WhatsAppPlaceholder from './pages/WhatsAppPlaceholder';
import SettingsPlaceholder from './pages/SettingsPlaceholder';
import AdminLayout from './layouts/AdminLayout';

function App() {
  const [isAuthenticated, setIsAuthenticated] = useState(() => {
    return sessionStorage.getItem('rc_auth') === 'true';
  });

  const handleLogin = () => {
    setIsAuthenticated(true);
    sessionStorage.setItem('rc_auth', 'true');
  };

  const handleLogout = () => {
    setIsAuthenticated(false);
    sessionStorage.removeItem('rc_auth');
  };

  return (
    <BrowserRouter>
      <Routes>
        {/* Public Login Route */}
        <Route
          path="/login"
          element={
            isAuthenticated ? <Navigate to="/dashboard" replace /> : <Login onLogin={handleLogin} />
          }
        />

        {/* Private Protected Routes */}
        <Route
          path="/dashboard"
          element={
            isAuthenticated ? (
              <AdminLayout onLogout={handleLogout}>
                <Dashboard />
              </AdminLayout>
            ) : (
              <Navigate to="/login" replace />
            )
          }
        />

        <Route
          path="/members"
          element={
            isAuthenticated ? (
              <AdminLayout onLogout={handleLogout}>
                <Members />
              </AdminLayout>
            ) : (
              <Navigate to="/login" replace />
            )
          }
        />

        <Route
          path="/chits"
          element={
            isAuthenticated ? (
              <AdminLayout onLogout={handleLogout}>
                <ChitsPlaceholder />
              </AdminLayout>
            ) : (
              <Navigate to="/login" replace />
            )
          }
        />

        <Route
          path="/payments"
          element={
            isAuthenticated ? (
              <AdminLayout onLogout={handleLogout}>
                <PaymentsPlaceholder />
              </AdminLayout>
            ) : (
              <Navigate to="/login" replace />
            )
          }
        />

        <Route
          path="/whatsapp"
          element={
            isAuthenticated ? (
              <AdminLayout onLogout={handleLogout}>
                <WhatsAppPlaceholder />
              </AdminLayout>
            ) : (
              <Navigate to="/login" replace />
            )
          }
        />

        <Route
          path="/settings"
          element={
            isAuthenticated ? (
              <AdminLayout onLogout={handleLogout}>
                <SettingsPlaceholder />
              </AdminLayout>
            ) : (
              <Navigate to="/login" replace />
            )
          }
        />
         
        {/* Default Fallback Redirects */}
        <Route
          path="*"
          element={<Navigate to={isAuthenticated ? "/dashboard" : "/login"} replace />}
        />
      </Routes>
    </BrowserRouter>
  );
}

export default App;
